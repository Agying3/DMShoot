原版确切样式（来自 08727b1 父提交 chat_view.py）

// 自己发的（橘色）
QLabel#bubble {
  background: rgba(255,150,50,0.18);
  border: 1px solid rgba(255,150,50,0.15);
  border-radius: 18px;
  color: #FFE0A0;
}
// 对方发的（磨砂白）
QLabel#bubble {
  background: rgba(255,255,255,0.07);
  border: 1px solid rgba(255,255,255,0.10);
  border-radius: 18px;
  color: #E2E2ED;
}
// 发件人昵称 / 时间
昵称 color: #4ADE80  (绿)
时间 color: rgba(255,255,255,0.50)
// ⚠️ 当时是单 QLabel 圆角矩形，没有「小尾巴」

// 现在（08727b1 起改成）
